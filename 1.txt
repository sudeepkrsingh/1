# PRACTICAL 5 : Statistical Language Model
# ─────────────────────────────────────────────────────────────────
print("\n"+SEP); print("PRACTICAL 5 : Unigram & Bigram Language Model + Add-1 Smoothing"); print(SEP)

lm_corpus = [
    "the cat sat on the mat",
    "the cat ate the rat",
    "the dog sat on the log",
    "a dog and a cat played in the park",
]

all_toks = []
for s in lm_corpus: all_toks.extend(s.split())

V         = len(set(all_toks))  # vocabulary size
N         = len(all_toks)
uni_freq  = Counter(all_toks)
bi_freq   = Counter(zip(all_toks, all_toks[1:]))

def p_unigram(w):       return (uni_freq[w]+1)/(N+V)
def p_bigram(w2,w1):    return (bi_freq[(w1,w2)]+1)/(uni_freq[w1]+V)

print(f"Corpus : {lm_corpus}")
print(f"Tokens : {N}  |  Vocabulary : {V}\n")

show_words = ['cat','dog','sat','rat','mat','played']
print(f"{'Word':<12} {'Count':>6}  {'P(w) smoothed':>16}")
print("-"*38)
for w in show_words:
    print(f"{w:<12} {uni_freq[w]:>6}  {p_unigram(w):>16.6f}")

print(f"\n{'Bigram':<18} {'Count':>6}  {'P(w2|w1) smoothed':>20}")
print("-"*48)
show_bigrams = [('cat','the'),('sat','cat'),('dog','the'),('log','on'),('park','the')]
for (w2,w1) in show_bigrams:
    print(f"P({w2}|{w1}):{'':<6} {bi_freq[(w1,w2)]:>3}  {p_bigram(w2,w1):>20.6f}")

test_sents = ["the cat sat on the mat", "the dog ate the rat", "a cat played in park"]
print("\nSentence Log-Probabilities (bigram, add-1 smoothing):")
for s in test_sents:
    wds  = s.split()
    logp = math.log(p_unigram(wds[0]))
    for i in range(1,len(wds)): logp += math.log(p_bigram(wds[i],wds[i-1]))
    print(f"  '{s}'  =>  log P = {logp:.4f}")

print("\n"+SEP); print("PRACTICAL 10 : Character-based Language Model (N-gram)"); print(SEP)

shakespeare = (
    "to be or not to be that is the question whether tis nobler in the mind "
    "to suffer the slings and arrows of outrageous fortune or to take arms "
    "against a sea of troubles and by opposing end them to die to sleep no "
    "more and by a sleep to say we end the heartache and the thousand natural "
    "shocks that flesh is heir to tis a consummation devoutly to be wished "
    "to die to sleep to sleep perchance to dream ay there is the rub for in "
    "that sleep of death what dreams may come when we have shuffled off this "
    "mortal coil must give us pause there is the respect that makes calamity "
    "of so long life for who would bear the whips and scorns of time "
)

print(f"Corpus: Shakespeare excerpt ({len(shakespeare)} characters)\n")

N_GRAM = 4

# Build n-gram model
model = defaultdict(Counter)
for i in range(len(shakespeare) - N_GRAM):
    ctx  = shakespeare[i:i+N_GRAM]
    nxt  = shakespeare[i+N_GRAM]
    model[ctx][nxt] += 1

vocab = sorted(set(shakespeare))
print(f"Vocabulary ({len(vocab)} unique chars): {repr(''.join(vocab))}")
print(f"N-gram order: {N_GRAM}")
print(f"Unique {N_GRAM}-gram contexts: {len(model)}\n")

def predict_next_char(ctx):
    ctx = ctx[-N_GRAM:]
    if ctx in model:
        return max(model[ctx], key=model[ctx].get)
    return ' '

# Show top contexts
print("Top 8 contexts and their most likely next character:")
top_ctx = sorted(model.items(), key=lambda x: -sum(x[1].values()))[:8]
for ctx, ctr in top_ctx:
    best = max(ctr, key=ctr.get)
    print(f"  '{ctx}'  =>  next: '{best}'  (top-3: {dict(ctr.most_common(3))})")

# Generate text
def generate_text(seed, length=80):
    text = seed
    for _ in range(length):
        text += predict_next_char(text)
    return text

seeds = ["to b", "the ", "slee", "drea"]
print("\nGenerated Text Samples:")
for seed in seeds:
    gen = generate_text(seed, 70)
    print(f"  Seed '{seed}': {gen}")

# Character-level prediction accuracy on corpus
correct = sum(predict_next_char(shakespeare[i:i+N_GRAM]) == shakespeare[i+N_GRAM]
              for i in range(len(shakespeare)-N_GRAM))
total   = len(shakespeare) - N_GRAM
print(f"\nCharacter Prediction Accuracy ({N_GRAM}-gram): {correct}/{total} = {correct/total*100:.1f}%")
