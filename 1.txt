# Practical 2: Password Cracking (MD5 Dictionary & Brute-force)
import hashlib, itertools, string

def md5(p): return hashlib.md5(p.encode()).hexdigest()

def dict_attack(target, wordlist):
    for w in wordlist:
        if md5(w) == target: return w
    return None

def brute_force(target, charset=string.ascii_lowercase, max_len=4):
    for length in range(1, max_len+1):
        for combo in itertools.product(charset, repeat=length):
            w = "".join(combo)
            if md5(w) == target: return w
    return None

def main():
    print("=== P2: Password Cracking ===")
    passwords = ["password", "admin", "hello", "abc"]
    wordlist = ["123456", "password", "admin", "hello", "letmein"]
    for p in passwords:
        h = md5(p)
        found = dict_attack(h, wordlist)
        print(f"  MD5({p!r}) = {h[:16]}...  Dict: {'FOUND: '+found if found else 'not found'}")
    # Brute-force short password
    target = md5("ab")
    print(f"\n  Brute-force 'ab': {brute_force(target)}")

if __name__ == "__main__": main()


# Practical 5: Caesar Cipher
def encrypt(text, shift):
    return "".join(chr((ord(c)-ord("A")+shift)%26+ord("A")) if c.isalpha() else c for c in text.upper())

def decrypt(text, shift): return encrypt(text, -shift)

def main():
    print("=== P5: Caesar Cipher ===")
    pt, shift = "Hello World", 3
    ct = encrypt(pt, shift)
    dt = decrypt(ct, shift)
    print(f"  Plaintext : {pt}")
    print(f"  Shift     : {shift}")
    print(f"  Encrypted : {ct}")
    print(f"  Decrypted : {dt}")
    print(f"  Result    : {'PASS' if dt.replace(' ','') == pt.upper().replace(' ','') else 'FAIL'}")
    print("\n  Brute-force:")
    for s in range(1, 26): print(f"    Shift {s:2d}: {decrypt(ct, s)}")

if __name__ == "__main__": main()
# Practical 6: Monoalphabetic & Vigenere Polyalphabetic Ciphers
import random, string

def mono_encrypt(text, key_map): return "".join(key_map.get(c, c) for c in text.upper())
def mono_decrypt(text, key_map):
    rev = {v:k for k,v in key_map.items()}; return "".join(rev.get(c,c) for c in text)

def vigenere(text, key, decrypt=False):
    result, ki = [], 0
    for c in text.upper():
        if c.isalpha():
            shift = ord(key[ki % len(key)]) - ord("A")
            if decrypt: shift = -shift
            result.append(chr((ord(c)-ord("A")+shift)%26+ord("A"))); ki += 1
        else: result.append(c)
    return "".join(result)

def main():
    print("=== P6: Monoalphabetic & Polyalphabetic Ciphers ===")
    alpha = list(string.ascii_uppercase)
    shuffled = alpha[:]; random.shuffle(shuffled)
    key_map = dict(zip(alpha, shuffled))
    pt = "HELLO WORLD"
    ct = mono_encrypt(pt, key_map); dt = mono_decrypt(ct, key_map)
    print(f"\n[Mono] Plaintext: {pt} -> Cipher: {ct} -> Decrypted: {dt}")
    print(f"  Result: {'PASS' if dt == pt else 'FAIL'}")
    pt2, key = "WEAREDISCOVERED", "DECEPTIVE"
    ct2 = vigenere(pt2, key); dt2 = vigenere(ct2, key, decrypt=True)
    print(f"\n[Vigenere] Plaintext: {pt2} Key: {key}")
    print(f"  Encrypted: {ct2}\n  Decrypted: {dt2}")
    print(f"  Result: {'PASS' if dt2 == pt2 else 'FAIL'}")

if __name__ == "__main__": main()
# Practical 7: Playfair Cipher
def make_matrix(key):
    seen, matrix = set(), []
    for c in (key+"ABCDEFGHIKLMNOPQRSTUVWXYZ"):
        c = c.upper().replace("J","I")
        if c.isalpha() and c not in seen: seen.add(c); matrix.append(c)
    return [matrix[i*5:(i+1)*5] for i in range(5)]

def find_pos(matrix, c):
    for r,row in enumerate(matrix):
        if c in row: return r, row.index(c)

def playfair_pair(matrix, a, b, enc=True):
    r1,c1 = find_pos(matrix,a); r2,c2 = find_pos(matrix,b)
    d = 1 if enc else -1
    if r1==r2: return matrix[r1][(c1+d)%5]+matrix[r2][(c2+d)%5]
    if c1==c2: return matrix[(r1+d)%5][c1]+matrix[(r2+d)%5][c2]
    return matrix[r1][c2]+matrix[r2][c1]

def prepare(text):
    text = text.upper().replace("J","I").replace(" ","")
    out = []
    i = 0
    while i < len(text):
        a = text[i]; b = text[i+1] if i+1<len(text) else "X"
        if a == b: b = "X"; i += 1
        else: i += 2
        out.append(a+b)
    return out

def playfair(text, key, enc=True):
    m = make_matrix(key); pairs = prepare(text)
    return "".join(playfair_pair(m, a, b, enc) for a,b in pairs)

def main():
    print("=== P7: Playfair Cipher ===")
    pt, key = "INSTRUMENTS", "MONARCHY"
    ct = playfair(pt, key); dt = playfair(ct, key, enc=False)
    print(f"  Plaintext : {pt}\n  Key       : {key}")
    print(f"  Encrypted : {ct}\n  Decrypted : {dt}")
    print(f"  Result    : {'PASS' if dt.rstrip('X') == pt.replace('J','I') else 'FAIL'}")

if __name__ == "__main__": main()
# Practical 8: Hill Cipher
import numpy as np

def mod_inv_matrix(matrix, mod=26):
    det = int(round(np.linalg.det(matrix))) % mod
    def ext_gcd(a,b):
        if b==0: return a,1,0
        g,x,y = ext_gcd(b,a%b); return g,y,x-(a//b)*y
    _,inv,_ = ext_gcd(det%mod, mod)
    inv %= mod
    n = matrix.shape[0]
    adj = np.array([[(-1)**(i+j)*round(np.linalg.det(np.delete(np.delete(matrix,i,0),j,1)))
                     for j in range(n)] for i in range(n)]).T % mod
    return (inv * adj) % mod

def hill(text, key_matrix, decrypt=False):
    n = key_matrix.shape[0]
    nums = [ord(c)-65 for c in text.upper() if c.isalpha()]
    while len(nums)%n: nums.append(23)  # pad with X
    K = mod_inv_matrix(key_matrix) if decrypt else key_matrix
    result = []
    for i in range(0, len(nums), n):
        v = np.array(nums[i:i+n])
        result.extend(int(x)%26 for x in K@v%26)
    return "".join(chr(x+65) for x in result)

def main():
    print("=== P8: Hill Cipher ===")
    K = np.array([[3,3],[2,5]])
    pt = "ACT"
    ct = hill(pt, K); dt = hill(ct, K, decrypt=True)
    print(f"  Plaintext : {pt}\n  Key Matrix: {K.tolist()}")
    print(f"  Encrypted : {ct}\n  Decrypted : {dt}")
    print(f"  Result    : {'PASS' if dt[:len(pt)] == pt else 'FAIL'}")

if __name__ == "__main__": main()
# Practical 9: Rail Fence Cipher
def rail_encrypt(text, rails):
    fence = [[] for _ in range(rails)]
    r, d = 0, 1
    for c in text:
        fence[r].append(c)
        if r==0: d=1
        elif r==rails-1: d=-1
        r+=d
    return "".join(c for row in fence for c in row)

def rail_decrypt(ct, rails):
    n = len(ct)
    pattern = []
    r, d = 0, 1
    for i in range(n):
        pattern.append(r)
        if r==0: d=1
        elif r==rails-1: d=-1
        r+=d
    idx = 0; pos = {}
    for rail in range(rails):
        for i,x in enumerate(pattern):
            if x==rail: pos[i]=ct[idx]; idx+=1
    return "".join(pos[i] for i in range(n))

def main():
    print("=== P9: Rail Fence Cipher ===")
    pt, rails = "WEAREDISCOVEREDFLEEATONCE", 3
    ct = rail_encrypt(pt, rails); dt = rail_decrypt(ct, rails)
    print(f"  Plaintext : {pt}\n  Rails     : {rails}")
    print(f"  Encrypted : {ct}\n  Decrypted : {dt}")
    print(f"  Result    : {'PASS' if dt==pt else 'FAIL'}")

if __name__ == "__main__": main()
# Practical 10: Row (Columnar) Transposition Cipher
def row_encrypt(text, key):
    text = text.upper().replace(" ","")
    n = len(key)
    padded = text.ljust(-(-len(text)//n)*n, "X")
    grid = [list(padded[i:i+n]) for i in range(0, len(padded), n)]
    order = sorted(range(n), key=lambda i: key[i])
    return "".join(grid[r][c] for c in order for r in range(len(grid)))

def row_decrypt(ct, key):
    n = len(key); rows = len(ct)//n
    order = sorted(range(n), key=lambda i: key[i])
    cols = {}
    idx = 0
    for c in order: cols[c]=list(ct[idx:idx+rows]); idx+=rows
    return "".join(cols[c][r] for r in range(rows) for c in range(n)).rstrip("X")

def main():
    print("=== P10: Row Transposition Cipher ===")
    pt, key = "ATTACK POSTPONED UNTIL TWO AM", [4,3,1,2,5,6,7]
    ct = row_encrypt(pt, key); dt = row_decrypt(ct, key)
    print(f"  Plaintext : {pt}\n  Key       : {key}")
    print(f"  Encrypted : {ct}\n  Decrypted : {dt}")
    print(f"  Result    : {'PASS' if dt == pt.upper().replace(' ','') else 'FAIL'}")

if __name__ == "__main__": main()
# Practical 11: Product Cipher (Caesar + Row Transposition)
def caesar(text, shift):
    return "".join(chr((ord(c)-65+shift)%26+65) if c.isalpha() else c for c in text.upper())

def row_enc(text, key):
    text = text.replace(" ",""); n = len(key)
    padded = text.ljust(-(-len(text)//n)*n, "X")
    grid = [list(padded[i:i+n]) for i in range(0, len(padded), n)]
    order = sorted(range(n), key=lambda i: key[i])
    return "".join(grid[r][c] for c in order for r in range(len(grid)))

def row_dec(ct, key):
    n = len(key); rows = len(ct)//n
    order = sorted(range(n), key=lambda i: key[i])
    cols = {}; idx = 0
    for c in order: cols[c]=list(ct[idx:idx+rows]); idx+=rows
    return "".join(cols[c][r] for r in range(rows) for c in range(n)).rstrip("X")

def product_enc(text, shift, key): return row_enc(caesar(text, shift), key)
def product_dec(text, shift, key): return caesar(row_dec(text, key), -shift)

def main():
    print("=== P11: Product Cipher ===")
    pt, shift, key = "HELLO WORLD", 3, [2,1,3]
    ct = product_enc(pt, shift, key); dt = product_dec(ct, shift, key)
    print(f"  Plaintext : {pt}\n  Caesar({shift}) : {caesar(pt, shift)}")
    print(f"  Encrypted : {ct}\n  Decrypted : {dt}")
    print(f"  Result    : {'PASS' if dt.replace('X','') == pt.upper().replace(' ','') else 'FAIL'}")
    print("\n  [Note] DES/AES are product ciphers: S-boxes (substitution) + P-boxes (permutation) repeated in rounds.")

if __name__ == "__main__": main()
